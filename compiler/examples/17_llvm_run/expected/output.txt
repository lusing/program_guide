; ModuleID = 'tip'
source_filename = "tip"

declare i32 @tip_input()

declare void @tip_output(i32)

define i32 @tip_main(i32 %0) {
entry:
  %n = alloca i32, align 4
  store i32 %0, ptr %n, align 4
  %f = alloca i32, align 4
  store i32 0, ptr %f, align 4
  store i32 1, ptr %f, align 4
  br label %wh.cond

wh.cond:                                          ; preds = %wh.body, %entry
  %n1 = load i32, ptr %n, align 4
  %1 = icmp sgt i32 %n1, 0
  %2 = zext i1 %1 to i32
  %3 = icmp ne i32 %2, 0
  br i1 %3, label %wh.body, label %wh.exit

wh.body:                                          ; preds = %wh.cond
  %f2 = load i32, ptr %f, align 4
  %n3 = load i32, ptr %n, align 4
  %4 = mul i32 %f2, %n3
  store i32 %4, ptr %f, align 4
  %n4 = load i32, ptr %n, align 4
  %5 = sub i32 %n4, 1
  store i32 %5, ptr %n, align 4
  br label %wh.cond

wh.exit:                                          ; preds = %wh.cond
  %f5 = load i32, ptr %f, align 4
  call void @tip_output(i32 %f5)
  %f6 = load i32, ptr %f, align 4
  ret i32 %f6
}

define i32 @tip_entry() {
entry:
  %0 = call i32 @tip_input()
  %1 = call i32 @tip_main(i32 %0)
  ret i32 %1
}
